type Array[T any] struct {
        data []T
}

func NewArray[T any]() *Array[T] {
        return &Array[T]{data: make([]T, 0)}
}

func (a *Array[T]) Push(val T) {
        a.data = append(a.data, val)
}

func (a *Array[T]) PushIdx(idx int, val T) error {
        if idx < 0 || idx > len(a.data) {
                return errors.New("index out of bounds")
        }
        a.data = append(a.data, *new(T))
        copy(a.data[idx+1:], a.data[idx:])
        a.data[idx] = val
        return nil
}

func (a *Array[T]) Get(idx int) (T, error) {
        var zero T
        if idx < 0 || idx >= len(a.data) {
                return zero, errors.New("index out of bounds")
        }
        return a.data[idx], nil
}

func (a *Array[T]) Set(idx int, val T) error {
        if idx < 0 || idx >= len(a.data) {
                return errors.New("index out of bounds")
        }
        a.data[idx] = val
        return nil
}

func (a *Array[T]) Del(idx int) error {
        if idx < 0 || idx >= len(a.data) {
                return errors.New("index out of bounds")
        }
        a.data = append(a.data[:idx], a.data[idx+1:]...)
        return nil
}

func (a *Array[T]) Len() int {
        return len(a.data)
}

func (a *Array[T]) Clear() {
        a.data = make([]T, 0)
}

func (a *Array[T]) SaveText(file string) error {
        data, err := json.Marshal(a.data)
        if err != nil {
                return err
        }
        return os.WriteFile(file, data, 0644)
}

func (a *Array[T]) LoadText(file string) error {
        data, err := os.ReadFile(file)
        if err != nil {
                return err
        }
        return json.Unmarshal(data, &a.data)
}

func (a *Array[T]) SaveBin(file string) error {
        f, err := os.Create(file)
        if err != nil {
                return err
        }
        defer f.Close()

        size := int32(len(a.data))
        binary.Write(f, binary.LittleEndian, size)

        for _, v := range a.data {
                val, ok := any(v).(int)
                if !ok {
                        return errors.New("binary serialization supports only int")
                }
                binary.Write(f, binary.LittleEndian, int32(val))
        }
        return nil
}

func (a *Array[T]) LoadBin(file string) error {
        f, err := os.Open(file)
        if err != nil {
                return err
        }
        defer f.Close()

        var size int32
        if err := binary.Read(f, binary.LittleEndian, &size); err != nil {
                return err
        }

        a.Clear()
        for i := int32(0); i < size; i++ {
                var val int32
                if err := binary.Read(f, binary.LittleEndian, &val); err != nil {
                        return err
                }

                var zero T
                valTyp := reflect.TypeOf(zero)
                if valTyp.Kind() == reflect.Int {
                        a.Push(any(int(val)).(T))
                } else if valTyp.Kind() == reflect.Int32 {
                        a.Push(any(val).(T))
                } else {
                        return errors.New("unsupported type for binary serialization")
                }
        }
        return nil
}

type hNode struct {
        Key   string
        Value string
        Next  *hNode
}

type HashTable struct {
        table [10]*hNode
}

type kvPair struct {
        Key   string `json:"key"`
        Value string `json:"value"`
}

func NewHashTable() *HashTable {
        return &HashTable{}
}

func (h *HashTable) hash(key string) int {
        var hash uint = 0
        for i := 0; i < len(key); i++ {
                hash = hash*31 + uint(key[i])
        }
        return int(hash % 10)
}

func (h *HashTable) Insert(key, value string) {
        idx := h.hash(key)
        for curr := h.table[idx]; curr != nil; curr = curr.Next {
                if curr.Key == key {
                        curr.Value = value
                        return
                }
        }
        h.table[idx] = &hNode{Key: key, Value: value, Next: h.table[idx]}
}

func (h *HashTable) Find(key string) (string, bool) {
        idx := h.hash(key)
        for curr := h.table[idx]; curr != nil; curr = curr.Next {
                if curr.Key == key {
                        return curr.Value, true
                }
        }
        return "", false
}

func (h *HashTable) Delete(key string) bool {
        idx := h.hash(key)
        var prev *hNode
        for curr := h.table[idx]; curr != nil; curr = curr.Next {
                if curr.Key == key {
                        if prev == nil {
                                h.table[idx] = curr.Next
                        } else {
                                prev.Next = curr.Next
                        }
                        return true
                }
                prev = curr
        }
        return false
}

func (h *HashTable) SaveText(file string) error {
        var pairs []kvPair
        for i := 0; i < 10; i++ {
                for curr := h.table[i]; curr != nil; curr = curr.Next {
                        pairs = append(pairs, kvPair{Key: curr.Key, Value: curr.Value})
                }
        }
        data, _ := json.Marshal(pairs)
        return os.WriteFile(file, data, 0644)
}

func (h *HashTable) LoadText(file string) error {
        data, err := os.ReadFile(file)
        if err != nil {
                return err
        }
        var pairs []kvPair
        if err := json.Unmarshal(data, &pairs); err != nil {
                return err
        }
        h.table = [10]*hNode{}
        for _, p := range pairs {
                h.Insert(p.Key, p.Value)
        }
        return nil
}

type fNode[T any] struct {
        Value T
        Next  *fNode[T]
}

type ForwardList[T any] struct {
        Head *fNode[T]
}

func NewForwardList[T any]() *ForwardList[T] {
        return &ForwardList[T]{}
}

func (l *ForwardList[T]) PushFront(val T) {
        l.Head = &fNode[T]{Value: val, Next: l.Head}
}

func (l *ForwardList[T]) PushBack(val T) {
        newNode := &fNode[T]{Value: val}
        if l.Head == nil {
                l.Head = newNode
                return
        }
        curr := l.Head
        for curr.Next != nil {
                curr = curr.Next
        }
        curr.Next = newNode
}

func (l *ForwardList[T]) DelFront() bool {
        if l.Head == nil {
                return false
        }
        l.Head = l.Head.Next
        return true
}

func (l *ForwardList[T]) SaveText(file string) error {
        var slice []T
        for curr := l.Head; curr != nil; curr = curr.Next {
                slice = append(slice, curr.Value)
        }
        data, _ := json.Marshal(slice)
        return os.WriteFile(file, data, 0644)
}

func (l *ForwardList[T]) LoadText(file string) error {
        data, err := os.ReadFile(file)
        if err != nil {
                return err
        }
        var slice []T
        if err := json.Unmarshal(data, &slice); err != nil {
                return err
        }
        l.Head = nil
        for _, v := range slice {
                l.PushBack(v)
        }
        return nil
}

type dNode[T any] struct {
        Value T
        Next  *dNode[T]
        Prev  *dNode[T]
}

type List[T any] struct {
        Head *dNode[T]
        Tail *dNode[T]
}

func NewList[T any]() *List[T] {
        return &List[T]{}
}

func (l *List[T]) PushFront(val T) {
        n := &dNode[T]{Value: val, Next: l.Head}
        if l.Head != nil {
                l.Head.Prev = n
        } else {
                l.Tail = n
        }
        l.Head = n
}

func (l *List[T]) PushBack(val T) {
        n := &dNode[T]{Value: val, Prev: l.Tail}
        if l.Tail != nil {
                l.Tail.Next = n
        } else {
                l.Head = n
        }
        l.Tail = n
}

func (l *List[T]) DelFront() bool {
        if l.Head == nil {
                return false
        }
        l.Head = l.Head.Next
        if l.Head != nil {
                l.Head.Prev = nil
        } else {
                l.Tail = nil
        }
        return true
}

func (l *List[T]) DelBack() bool {
        if l.Tail == nil {
                return false
        }
        l.Tail = l.Tail.Prev
        if l.Tail != nil {
                l.Tail.Next = nil
        } else {
                l.Head = nil
        }
        return true
}

func (l *List[T]) SaveText(file string) error {
        var slice []T
        for curr := l.Head; curr != nil; curr = curr.Next {
                slice = append(slice, curr.Value)
        }
        data, _ := json.Marshal(slice)
        return os.WriteFile(file, data, 0644)
}

func (l *List[T]) LoadText(file string) error {
        data, err := os.ReadFile(file)
        if err != nil {
                return err
        }
        var slice []T
        if err := json.Unmarshal(data, &slice); err != nil {
                return err
        }
        l.Head, l.Tail = nil, nil
        for _, v := range slice {
                l.PushBack(v)
        }
        return nil
}

type tNode struct {
        Value int
        Left  *tNode
        Right *tNode
}

type CompleteBinaryTree struct {
        Root *tNode
        Size int
}

func NewTree() *CompleteBinaryTree {
        return &CompleteBinaryTree{}
}

func (t *CompleteBinaryTree) getNode(index int) *tNode {
        if index < 1 || t.Root == nil {
                return nil
        }
        curr := t.Root
        mask := 1
        for mask <= index/2 {
                mask *= 2
        }
        mask /= 2
        for mask > 0 && curr != nil {
                if (index & mask) != 0 {
                        curr = curr.Right
                } else {
                        curr = curr.Left
                }
                mask /= 2
        }
        return curr
}

func (t *CompleteBinaryTree) Insert(val int) {
        n := &tNode{Value: val}
        if t.Size == 0 {
                t.Root = n
        } else {
                parent := t.getNode((t.Size + 1) / 2)
                if (t.Size+1)%2 == 0 {
                        parent.Left = n
                } else {
                        parent.Right = n
                }
        }
        t.Size++
}

func (t *CompleteBinaryTree) Find(val int) bool {
        for i := 1; i <= t.Size; i++ {
                if n := t.getNode(i); n != nil && n.Value == val {
                        return true
                }
        }
        return false
}

func (t *CompleteBinaryTree) SaveText(file string) error {
        var slice []int
        for i := 1; i <= t.Size; i++ {
                if n := t.getNode(i); n != nil {
                        slice = append(slice, n.Value)
                }
        }
        data, _ := json.Marshal(slice)
        return os.WriteFile(file, data, 0644)
}

func (t *CompleteBinaryTree) LoadText(file string) error {
        data, err := os.ReadFile(file)
        if err != nil {
                return err
        }
        var slice []int
        if err := json.Unmarshal(data, &slice); err != nil {
                return err
        }
        t.Root, t.Size = nil, 0
        for _, v := range slice {
                t.Insert(v)
        }
        return nil
}
